# ============================
#   System Status Dashboard
# ============================

Clear-Host
Write-Host "==============================" -ForegroundColor Cyan
Write-Host "   SYSTEM STATUS DASHBOARD    " -ForegroundColor Yellow
Write-Host "==============================" -ForegroundColor Cyan

# --- CPU ---
$cpu = Get-CimInstance Win32_Processor
Write-Host "`n[CPU]" -ForegroundColor Green
Write-Host ("Name: {0}" -f $cpu.Name)
Write-Host ("Load: {0}%" -f $cpu.LoadPercentage) -ForegroundColor Magenta

# --- MEMORY ---
$os = Get-CimInstance Win32_OperatingSystem
$total = [math]::Round($os.TotalVisibleMemorySize / 1MB, 2)
$free  = [math]::Round($os.FreePhysicalMemory / 1MB, 2)
$used  = [math]::Round($total - $free, 2)

Write-Host "`n[MEMORY]" -ForegroundColor Green
Write-Host ("Total: {0} GB" -f $total)
Write-Host ("Used : {0} GB" -f $used) -ForegroundColor Magenta
Write-Host ("Free : {0} GB" -f $free)

# --- DISK ---
$disk = Get-CimInstance Win32_LogicalDisk -Filter "DriveType=3"
Write-Host "`n[DISK]" -ForegroundColor Green
foreach ($d in $disk) {
    $size = [math]::Round($d.Size / 1GB, 2)
    $free = [math]::Round($d.FreeSpace / 1GB, 2)
    $used = [math]::Round($size - $free, 2)
    Write-Host ("Drive {0}: {1} GB total | {2} GB used | {3} GB free" -f $d.DeviceID, $size, $used, $free) -ForegroundColor Cyan
}

# --- SYSTEM INFO ---
$info = Get-ComputerInfo
Write-Host "`n[SYSTEM]" -ForegroundColor Green
Write-Host ("Computer Name: {0}" -f $info.CsName)
Write-Host ("OS: {0}" -f $info.OsName)
Write-Host ("Version: {0}" -f $info.OsVersion)
Write-Host ("Logical CPUs: {0}" -f $info.CsNumberOfLogicalProcessors)

Write-Host "`n==============================" -ForegroundColor Cyan
Write-Host "       STATUS COMPLETE         " -ForegroundColor Yellow
Write-Host "==============================" -ForegroundColor Cyan

# Show CPU Usage
Get-CimInstance Win32_Processor | Select-Object Name, LoadPercentage

# Show Memory Usage
Get-CimInstance Win32_OperatingSystem |
Select-Object FreePhysicalMemory, TotalVisibleMemorySize

# To show it in GB and percentage:
$os = Get-CimInstance Win32_OperatingSystem
$total = $os.TotalVisibleMemorySize / 1MB
$free  = $os.FreePhysicalMemory / 1MB
$used  = $total - $free

[PSCustomObject]@{
    TotalGB = "{0:N2}" -f $total
    UsedGB  = "{0:N2}" -f $used
    FreeGB  = "{0:N2}" -f $free
}

# Show Overall System Status
Get-ComputerInfo | Select-Object CsName, OsName, OsVersion, WindowsProductName, WindowsEditionId, CsNumberOfLogicalProcessors

# Or a simpler summary:
Get-ComputerInfo -Property Cs*,Os*

# ⚡ Combined Quick Status Script
Write-Host "=== CPU ==="
Get-CimInstance Win32_Processor | Select Name, LoadPercentage

Write-Host "`n=== MEMORY ==="
$os = Get-CimInstance Win32_OperatingSystem
$total = $os.TotalVisibleMemorySize / 1MB
$free  = $os.FreePhysicalMemory / 1MB
$used  = $total - $free
"{0:N2} GB Total | {1:N2} GB Used | {2:N2} GB Free" -f $total, $used, $free

Write-Host "`n=== SYSTEM ==="
Get-ComputerInfo | Select CsName, OsName, OsVersion
